/**
 * Purchases an item from the magic shop.
 *
 * @param itemName the name of the item to purchase (case-sensitive)
 * @param bankAccount the player's bank account for debit
 * @return Optional containing true if purchase succeeded, empty if failed
 *
 * @pre itemName != null && !itemName.trim().isEmpty()
 * @pre bankAccount != null
 * @pre catalog.containsKey(itemName)
 * @pre bankAccount.getBalance().compareTo(catalog.get(itemName).getPrice()) >= 0
 *
 * @post If successful: playerInventory contains the new item
 * @post If successful: bankAccount balance decreased by item price
 * @post No side effects if purchase fails
 */
public Optional<Boolean> buyItem(String itemName, BankAccount bankAccount) {
    if (itemName == null || itemName.trim().isEmpty()) {
        System.err.println("Precondition violated: itemName cannot be null or empty");
        return Optional.empty();
    }
    
    if (bankAccount == null) {
        System.err.println("Precondition violated: bankAccount cannot be null");
        return Optional.empty();
    }
    
    Item item = catalog.get(itemName);
    if (item == null) {
        System.out.println("Item not available: " + itemName);
        return Optional.empty();
    }
    
    BigDecimal price = item.getPrice();
    
    if (bankAccount.withdraw(price)) {
        playerInventory.add(item);
        System.out.printf("✓ Purchased %s for R$ %.2f%n", itemName, price);
        return Optional.of(true);
    }
    
    return Optional.empty();
}

/**
 * Compra um item da loja, debitando o valor da conta bancária do jogador.

public boolean buyItem(String itemName, BankAccount bankAccount) {
    if (itemName == null || bankAccount == null) {
        System.out.println("Invalid transaction.");
        return false;
    }
    
    if (!itemsForSale.containsKey(itemName)) {
        System.out.println("Item not found in shop: " + itemName);
        return false;

    double price = itemsForSale.get(itemName);
    
    if (bankAccount.withdraw(price)) {
        playerInventory.add(new Item(itemName));
        System.out.printf("You bought %s for R$ %.2f!\n", itemName, price);
        return true;
    }
    
    return false;
}

public Optional<Boolean> sellItem(String itemName, BankAccount bankAccount) {
    if (itemName == null || itemName.trim().isEmpty()) {
        return Optional.empty();
    }
    
    // Busca case-insensitive (melhor UX)
    Item itemToSell = playerInventory.stream()
        .filter(item -> item.getName().equalsIgnoreCase(itemName))
        .findFirst()
        .orElse(null);
    
    if (itemToSell == null) {
        System.out.println("Item not found in inventory: " + itemName);
        return Optional.empty();
    }
    
    Item catalogItem = catalog.get(itemToSell.getName());
    if (catalogItem == null) {
        System.out.println("Shop does not buy this item.");
        return Optional.empty();
    }
    
    BigDecimal sellPrice = catalogItem.getPrice()
        .multiply(DEPRECIATION_RATE)
        .setScale(2, RoundingMode.HALF_EVEN);
    
    playerInventory.remove(itemToSell);
    bankAccount.deposit(sellPrice);
    
    System.out.printf("✓ Sold %s for R$ %.2f (50%% depreciation)%n", 
                      itemName, sellPrice);
    return Optional.of(true);
}

public boolean sellItem(String itemName, BankAccount bankAccount) {
    if (itemName == null || bankAccount == null) {
        System.out.println("Invalid transaction.");
        return false;
    }
    
    Item itemToSell = null;
    for (Item item : playerInventory) {
        if (item.getName().equalsIgnoreCase(itemName)) {
            itemToSell = item;
            break;
        }
    }
    
    if (itemToSell == null) {
        System.out.println("You don't have that item: " + itemName);
        return false;
    }
    
    if (!itemsForSale.containsKey(itemName)) {
        System.out.println("Shop doesn't buy this item.");
        return false;
    }
    
    double originalPrice = itemsForSale.get(itemName);
    double sellPrice = originalPrice * 0.5;
    
    playerInventory.remove(itemToSell);
    bankAccount.deposit(sellPrice);
    System.out.printf("You sold %s for R$ %.2f (50%% of original price).\n", 
                      itemName, sellPrice);
    return true;
}

import java.math.BigDecimal;
import java.math.RoundingMode;

public class BankAccount {
    private BigDecimal balance;
    
    public BankAccount() {
        this.balance = new BigDecimal("1000.00");
    }
    
    public boolean deposit(BigDecimal amount) {
        if (amount == null || amount.compareTo(BigDecimal.ZERO) <= 0) {
            return false;
        }
        balance = balance.add(amount);
        return true;
    }
    
    public boolean withdraw(BigDecimal amount) {
        if (amount == null || amount.compareTo(BigDecimal.ZERO) <= 0) {
            return false;
        }
        if (amount.compareTo(balance) > 0) {
            return false;
        }
        balance = balance.subtract(amount);
        return true;
    }
}

CÓDIGO IMPLEMENTADO:
--------------------------------------------------
public class BankAccount {
    private double balance;
    
    public BankAccount() {
        this.balance = 1000.0;
    }
    
    public boolean deposit(double amount) {
        if (amount <= 0) {
            System.out.println("Invalid deposit amount.");
            return false;
        }
        this.balance += amount;
        return true;
    }
}

PROBLEMA GRAVE NO NOSSO CÓDIGO:
--------------------------------------------------
0.1 + 0.2 = 0.30000000000000004  // ❌ Erro de precisão com double!

Com BigDecimal:
new BigDecimal("0.1").add(new BigDecimal("0.2")) = 0.30  // ✅ Correto!

PONTOS FORTES DO CÓDIGO DA IA (DeepSeek):
--------------------------------------------------
1. Uso de BigDecimal para dinheiro (evita erros de arredondamento)
2. Separação de responsabilidades (UI vs lógica de negócio)

4. Encapsulamento real (listas imutáveis, final)
5. Notação formal nas pré/pós-condições
6. Uso de Stream API para código mais expressivo
7. Constantes para valores mágicos (DEPRECIATION_RATE)
